176 const auto dy = size_y_ / ny;
177 const auto dz = size_z_ / nz;
178 const auto dV = dx * dy * dz;

180 for (int i = 0; i < nx; ++i) {
181 for (int j = 0; j < ny; ++j) {
182 for (int k = 0; k < nz; ++k) {
183 const auto x = (i + 0.5) / nx * size_x_ - size_x_ / 2;
184 const auto y = (j + 0.5) / ny * size_y_ - size_y_ / 2;
185 const auto z = (k + 0.5) / nz * size_z_ - size_z_ / 2;
