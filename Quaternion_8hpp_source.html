379 static auto Slerp(const Quaternion &q1, const Quaternion<U> &q2, const Scalar t) {
380 auto cos_angle = q1.Dot(q2);
381 auto q2_adjusted = q2;
382 if (cos_angle < 0) {
383 q2_adjusted = -q2_adjusted;
384 cos_angle = -cos_angle;
385 }
386
387 const auto cos_angle_clamped = Utils::clamp(cos_angle, -1_, 1_);
388 const auto angle = Utils::acos(cos_angle_clamped);
389
390 if (angle < 1e-6_rad) {
391 return (q1 * (1_ - t) + q2_adjusted * t).Normalized();
392 }
393
394 const auto sin_angle = Utils::sin(angle);
395 const auto a = Utils::sin((1_ - t) * angle) / sin_angle;
396 const auto b = Utils::sin(t * angle) / sin_angle;
397
398 return (q1 * a + q2_adjusted * b).Normalized();
399 }
