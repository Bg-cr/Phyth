94 [[nodiscard]] Vector3<Quantity<NewtonPerCoulomb> >
95 GetElectricFieldAt(const Vector3<Quantity<Meter> > &point) const override {
96 const auto r = point - position_;
97 if (r.Length() < Quantity<Meter>(Config::epsilon)) {
98 throw std::runtime_error("PointCharge: field diverges at charge location");
99 }
100
101 if (fixed_) {
102 return Consts::k_E * charge_value_ / r.LengthSquared() * r.Normalized();
103 }
104 Quantity<Second> retarded_dt;
105 try {
106 retarded_dt = SolveRetardedTimeOffset(point);
107 } catch (std::runtime_error &) {
108 Vector3<Quantity<Meter> > retarded_r = r;
109 if (!position_history_.IsEmpty()) {
110 retarded_r = position_history_.GetValueByTime(0_s);
111 // The observation point has not yet seen the movement of charges.
112 // It is an electrostatic field of charges at p_0.
113 }
114 return Consts::k_E * charge_value_ / retarded_r.LengthSquared() * retarded_r.Normalized();
115 }
116
117 const auto retarded_pos = GetPositionAtOffset(retarded_dt);
118 const auto retarded_v = GetVelocityAtOffset(retarded_dt);
119 const auto retarded_a = GetAccelerationAtOffset(retarded_dt);
120
121 const auto R = point - retarded_pos;
122 const auto hat_R = R.Normalized();
123 const auto beta = retarded_v / Consts::c;
124 const auto beta_a = retarded_a / Consts::c;
125 const auto gamma = 1 / Utils::sqrt(1_ - Utils::square(beta.Length()));
126
127 const auto velocity_field = Consts::k_E * charge_value_ * (hat_R - beta)
128 / Utils::square(gamma * R.Length())
129 / Utils::cube(1_ - hat_R.Dot(beta));
130
131 const auto acceleration_field = Consts::k_E * charge_value_
132 * hat_R.Cross((hat_R - beta).Cross(beta_a))
133 / Consts::c
134 / Utils::cube(1_ - hat_R.Dot(beta))
135 / R.Length();
136 return velocity_field + acceleration_field;
137 }

150 [[nodiscard]] Vector3<Quantity<Tesla> >
151 GetMagneticFieldAt(const Vector3<Quantity<Meter> > &point) const override {
152 if (fixed_) {
153 return {};
154 }
155 const auto r = point - position_;
156 Quantity<Second> retarded_dt;
157 try {
158 retarded_dt = SolveRetardedTimeOffset(point);
159 } catch (std::runtime_error &) {
160 return Consts::mu_0 / Consts::varpi * charge_value_ * velocity_.Cross(r.Normalized()) / r.
161 LengthSquared();
162 }
163 const auto R = point - GetPositionAtOffset(retarded_dt);
164 const auto hat_R = R.Normalized();
165
166 return (hat_R / Consts::c).Cross(GetElectricFieldAt(point));
167 }

181 [[nodiscard]] Quantity<Volt>
182 GetElectricPotentialAt(const Vector3<Quantity<Meter> > &point) const override {
183 const auto r = point - position_;
184 const auto dist = r.Length();
185
186 if (dist < Quantity<Meter>(Config::epsilon)) {
187 throw std::runtime_error("PointCharge: potential diverges at charge location");
188 }
189
190 if (fixed_ || position_history_.IsEmpty()) {
191 return Consts::k_E * charge_value_ / dist;
192 }
193
194 Quantity<Second> retarded_dt;
195 try {
196 retarded_dt = SolveRetardedTimeOffset(point);
197 } catch (std::runtime_error &) {
198 return Consts::k_E * charge_value_ / dist;
199 }
200
201 const auto retarded_pos = GetPositionAtOffset(retarded_dt);
202 const auto retarded_v = GetVelocityAtOffset(retarded_dt);
203
204 const auto R_vec = point - retarded_pos;
205 const auto R = R_vec.Length();
206 const auto hat_R = R_vec.Normalized();
207 const auto beta = retarded_v / Consts::c;
208
209 const auto denominator = R - hat_R.Dot(beta) * R;
210
211 if (denominator < Quantity<Meter>(Config::epsilon)) {
212 throw std::runtime_error("PointCharge: line-of-sight aligns with charge velocity at light speed");
213 }
214
215 return Consts::k_E * charge_value_ / denominator;
216 }

322 [[nodiscard]] Quantity<Second>
323 SolveRetardedTimeOffset(const Vector3<Quantity<Meter> > &point) const {
324 if (position_history_.IsEmpty()) {
325 return 0_s;
326 }
327
328 const Quantity<Meter> R0 = (point - position_).Length();
329 Quantity<Second> dt_delay = R0 / Consts::c;
330
331 const Quantity<Second> max_dt = static_cast<double>(position_history_.GetSize())
332 * Config::dt;
333 dt_delay = std::clamp(dt_delay, Quantity<Second>(0.0), max_dt);
334
335 for (int iter = 0; iter < Config::max_iterations; ++iter) {
336 const Vector3<Quantity<Meter> > pos_r = GetPositionAtOffset(dt_delay);
337 const Vector3<Quantity<MeterPerSecond> > v_r = GetVelocityAtOffset(dt_delay);
338
339 const Vector3<Quantity<Meter> > R_vec = point - pos_r;
340 const Quantity<Meter> R = R_vec.Length();
341 const Vector3<Scalar> hat_R = R_vec.Normalized();
342
343 const Quantity<Meter> f = R - Consts::c * dt_delay;
344
345 if (Utils::abs(f / Consts::c) < Quantity<Second>(Config::tolerance)) {
346 return dt_delay;
347 }
348
349 const Quantity<MeterPerSecond> df = -hat_R.Dot(v_r) - Consts::c;
350
351 if (Utils::abs(df) < Quantity<MeterPerSecond>(Config::epsilon)) {
352 break;
353 }
354
355 const Quantity<Second> delta = f / df;
356 dt_delay = dt_delay - delta;
357 dt_delay = std::clamp(dt_delay, Quantity<Second>(0.0), max_dt);
358 }
359
360 return SolveRetardedTimeOffsetBisection(point, Quantity<Second>(0.0), max_dt);
361 }

370 [[nodiscard]] Quantity<Second>
371 SolveRetardedTimeOffsetBisection(const Vector3<Quantity<Meter> > &point,
372 Quantity<Second> dt_low,
373 Quantity<Second> dt_high) const {
374 auto ComputeResidual = [&](const Quantity<Second> dt) -> Quantity<Meter> {
375 const Vector3<Quantity<Meter> > pos_r = GetPositionAtOffset(dt);
376 const Quantity<Meter> R = (point - pos_r).Length();
377 return R - Consts::c * dt;
378 };
379
380 Quantity<Meter> f_low = ComputeResidual(dt_low);
381 Quantity<Meter> f_high = ComputeResidual(dt_high);
382
383 if (f_low * f_high > 0_m2) {
384 throw std::runtime_error(
385 "PointCharge: signal has not reached observation point or no solution in interval");
386 }
387
388 for (int iter = 0; iter < Config::max_iterations; ++iter) {
389 const Quantity<Second> dt_mid = (dt_low + dt_high) / 2;
390 const Quantity<Meter> f_mid = ComputeResidual(dt_mid);
391
392 if (Utils::abs(f_mid / Consts::c) < Quantity<Second>(Config::tolerance)) {
393 return dt_mid;
394 }
395
396 if (f_low * f_mid < 0_m2) {
397 dt_high = dt_mid;
398 f_high = f_mid;
399 } else {
400 dt_low = dt_mid;
401 f_low = f_mid;
402 }
403 }
404
405 throw std::runtime_error("PointCharge: bisection failed to converge");
406 }
407 };
