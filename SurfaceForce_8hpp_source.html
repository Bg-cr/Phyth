155 const auto prefactor = -1.0 / (8.0 * Consts::pi * (1.0 - poisson_ratio));
156
157 const auto du = width_ / nu;
158 const auto dv = height_ / nv;
159 const auto dA = du * dv;
160 const auto dF = traction_ * dA;
161
162 for (int i = 0; i < nu; ++i) {
163 for (int j = 0; j < nv; ++j) {
164 const auto u = (i + 0.5) / nu * width_ - width_ / 2.0;
165 const auto v = (j + 0.5) / nv * height_ - height_ / 2.0;
166
167 const auto pos = center_ + u * u_axis_ + v * v_axis_;
168 const auto R = point - pos;
169 const auto R_mag = R.Length();
170
171 if (R_mag < Quantity<Meter>(Config::epsilon)) {
172 continue;
173 }
174
175 const auto R3 = R_mag * R_mag * R_mag;
176 const auto R5 = R3 * R_mag * R_mag;
177
178 const auto one_minus_2nu = 1.0 - 2.0 * poisson_ratio;
179 const auto dF_dot_R = dF.Dot(R);
180
181 for (int m = 0; m < 3; ++m) {
182 for (int n = 0; n < 3; ++n) {
183 const auto term1 = one_minus_2nu * (
184 dF[m] * R[n] + dF[n] * R[m]
185 - (m == n ? 1.0 : 0.0) * dF_dot_R
186 ) / R3;
187
188 const auto term2 = 3.0 * dF_dot_R * R[m] * R[n] / R5;
189
190 stress(m, n) += prefactor * (term1 - term2);
191 }
192 }
193 }
