175 for (int j = 0; j < ny; ++j) {
176 for (int k = 0; k < nz; ++k) {
177 const auto x = (i + 0.5) / nx * width_ - half_w;
178 const auto y = (j + 0.5) / ny * height_ - half_h;
179 const auto z = (k + 0.5) / nz * depth_ - half_d;
180
181 const auto pos = center_ + Vector3(x, y, z);
182 const auto R = point - pos;
183 const auto R_mag = R.Length();
184
185 if (R_mag < Quantity<Meter>(Config::epsilon)) {
186 continue;
187 }
188
189 const auto R3 = R_mag * R_mag * R_mag;
190 const auto R5 = R3 * R_mag * R_mag;
191
192 const auto one_minus_2nu = 1.0 - 2.0 * poisson_ratio;
193 const auto dF_dot_R = dF.Dot(R);
194
195 for (int m = 0; m < 3; ++m) {
196 for (int n = 0; n < 3; ++n) {
197 const auto term1 = one_minus_2nu * (
198 dF[m] * R[n] + dF[n] * R[m]
199 - (m == n ? 1.0 : 0.0) * dF_dot_R
200 ) / R3;

202 const auto term2 = 3.0 * dF_dot_R * R[m] * R[n] / R5;
