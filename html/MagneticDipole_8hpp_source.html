101 GetMagneticFieldAt(const Vector3<Quantity<Meter> > &point) const override {
102 const auto r = point - position_;
103 const auto dist = r.Length();
104
105 if (dist < Quantity<Meter>(Config::epsilon)) {
106 throw std::runtime_error("MagneticDipole: field diverges at dipole location");
107 }
108
109 if (fixed_ || position_history_.GetSize() < 4) {
110 const auto hat_r = r / dist;
111 constexpr auto coeff = Consts::mu_0 / (4 * Consts::pi);
112 return coeff * (3 * moment_.Dot(hat_r) * hat_r - moment_) / Utils::cube(dist);
113 }
114
115 try {
116 const Quantity<Second> retarded_dt = SolveRetardedTimeOffset(point);
117 const auto retarded_pos = GetPositionAtOffset(retarded_dt);
118 const auto retarded_m = GetMomentAtOffset(retarded_dt);
119 const auto retarded_m_dot = ComputeMomentFirstDerivative(retarded_dt);
120 const auto retarded_m_ddot = ComputeMomentSecondDerivative(retarded_dt);
121
122 const auto R = point - retarded_pos;
123 const auto R_mag = R.Length();
124 if (R_mag < Quantity<Meter>(Config::epsilon)) {
125 throw std::runtime_error("MagneticDipole: field diverges at dipole location");
126 }
127 const auto hat_R = R / R_mag;
128 constexpr auto coeff = Consts::mu_0 / (4 * Consts::pi);
129
130 const auto term1 = coeff * (3 * retarded_m.Dot(hat_R) * hat_R - retarded_m) / Utils::cube(R_mag);
131 const auto term2 = coeff * (3 * retarded_m_dot.Dot(hat_R) * hat_R - retarded_m_dot) /
132 (Consts::c * R_mag * R_mag);
133 const auto term3 = coeff * retarded_m_ddot.Cross(hat_R).Cross(hat_R) / (Consts::c * Consts::c * R_mag);
134
135 return term1 + term2 + term3;
136 } catch (const std::runtime_error &) {
137 const auto hat_r = r / dist;
138 constexpr auto coeff = Consts::mu_0 / (4 * Consts::pi);
139 return coeff * (3 * moment_.Dot(hat_r) * hat_r - moment_) / Utils::cube(dist);

219 GetElectricPotentialAt(const Vector3<Quantity<Meter> > &point) const override {
220 if ((point - position_).Length() < Quantity<Meter>(Config::epsilon)) {
221 throw std::runtime_error("MagneticDipole: potential diverges at dipole location");
222 }
223
224 if (fixed_ || position_history_.GetSize() < 4 || velocity_.Length() < Quantity<MeterPerSecond>(
225 Config::epsilon)) {

228
229 try {
230 const Quantity<Second> retarded_dt = SolveRetardedTimeOffset(point);
231 const auto retarded_pos = GetPositionAtOffset(retarded_dt);
232 const auto retarded_v = GetVelocityAtOffset(retarded_dt);
233 const auto retarded_p = GetElectricDipoleAtOffset(retarded_dt);
234
235 const auto R_vec = point - retarded_pos;
236 const auto R = R_vec.Length();
237 const auto hat_R = R_vec.Normalized();

240 const auto denominator_factor = Scalar(1) - hat_R.Dot(beta);

243 throw std::runtime_error("MagneticDipole: line-of-sight aligns with velocity at light speed");
244 }
245
246 return Consts::k_E * retarded_p.Dot(hat_R) / (R * R * denominator_factor);
247 } catch (const std::runtime_error &) {
248 return 0_V;

480 [[nodiscard]] static Vector3<Quantity<CoulombMeter> >
481 ComputeRelativisticDipoleMoment(
482 const Vector3<Quantity<MeterPerSecond> > &v,
483 const Vector3<Quantity<AmpereMeterSquared> > &m
484 ) {
485 const auto beta = v / Consts::c;
486 const auto beta2 = beta.LengthSquared();
487 if (beta2 > Scalar(1 - Config::epsilon)) {
488 return -v.Cross(m) / (Consts::c * Consts::c);
489 }
490 const auto gamma = Scalar(1) / Utils::sqrt(Scalar(1) - beta2);
491 return -gamma * v.Cross(m) / (Consts::c * Consts::c);
492 }

541 [[nodiscard]] Quantity<Second>
542 SolveRetardedTimeOffset(const Vector3<Quantity<Meter> > &point) const {
543 if (position_history_.IsEmpty()) {
544 return 0_s;
545 }
546
547 const Quantity<Meter> R0 = (point - position_).Length();
548 Quantity<Second> dt_delay = R0 / Consts::c;
549
550 const Quantity<Second> max_dt = static_cast<double>(position_history_.GetSize())
551 * Config::dt;
552 dt_delay = std::clamp(dt_delay, Quantity<Second>(0.0), max_dt);
553
554 for (int iter = 0; iter < Config::max_iterations; ++iter) {
555 const auto pos_r = GetPositionAtOffset(dt_delay);
556 const auto v_r = GetVelocityAtOffset(dt_delay);
557
558 const Vector3<Quantity<Meter> > R_vec = point - pos_r;
559 const Quantity<Meter> R = R_vec.Length();
560 const Vector3<Scalar> hat_R = R_vec.Normalized();
561
562 const Quantity<Meter> f = R - Consts::c * dt_delay;
563
564 if (Utils::abs(f / Consts::c) < Quantity<Second>(Config::tolerance)) {
565 return dt_delay;
566 }
567
568 const Quantity<MeterPerSecond> df = -hat_R.Dot(v_r) - Consts::c;
569
570 if (Utils::abs(df) < Quantity<MeterPerSecond>(Config::epsilon)) {
571 break;
572 }
573
574 const Quantity<Second> delta = f / df;
575 dt_delay = dt_delay - delta;
576 dt_delay = std::clamp(dt_delay, Quantity<Second>(0.0), max_dt);
577 }
578
579 return SolveRetardedTimeOffsetBisection(point, Quantity<Second>(0.0), max_dt);
580 }

587 [[nodiscard]] Quantity<Second>
588 SolveRetardedTimeOffsetBisection(const Vector3<Quantity<Meter> > &point,
589 Quantity<Second> dt_low,
590 Quantity<Second> dt_high) const {
591 auto ComputeResidual = [&](const Quantity<Second> dt) -> Quantity<Meter> {
592 const auto pos_r = GetPositionAtOffset(dt);
593 const Quantity<Meter> R = (point - pos_r).Length();
594 return R - Consts::c * Config::dt;
595 };
596
597 Quantity<Meter> f_low = ComputeResidual(dt_low);
598 Quantity<Meter> f_high = ComputeResidual(dt_high);
599
600 if (f_low * f_high > 0_m2) {
601 return Utils::abs(f_low) < Utils::abs(f_high) ? dt_low : dt_high;
602 }
603
604 for (int iter = 0; iter < Config::max_iterations; ++iter) {
605 const Quantity<Second> dt_mid = (dt_low + dt_high) / 2;
606 const Quantity<Meter> f_mid = ComputeResidual(dt_mid);
607
608 if (Utils::abs(f_mid / Consts::c) < Quantity<Second>(Config::tolerance)) {
609 return dt_mid;
610 }
611
612 if (f_low * f_mid < 0_m2) {
613 dt_high = dt_mid;
614 f_high = f_mid;
615 } else {
616 dt_low = dt_mid;
617 f_low = f_mid;
618 }
619 }
620
621 return (dt_low + dt_high) / 2;
622 }

763 ComputeMagneticGradientForce(const MagneticSource &field_source) const {

765 const auto m = moment_;
766
767 Vector3<Quantity<Newton> > force;
768
769 auto pos_x = position_;
770 pos_x.x += eps_m;

772 pos_x_neg.x -= eps_m;
773 const auto dB_dx = (field_source.GetMagneticFieldAt(pos_x) -
774 field_source.GetMagneticFieldAt(pos_x_neg)) / (2 * eps_m);

783 force.y = m.Dot(dB_dy);

786 pos_z.z += eps_m;

788 pos_z_neg.z -= eps_m;

808 Vector3<Quantity<Newton> > force;
809
810 auto pos_x = position_;
811 pos_x.x += eps_m;
812 auto pos_x_neg = position_;
813 pos_x_neg.x -= eps_m;
814 const auto dE_dx = (field_source.GetElectricFieldAt(pos_x) -
815 field_source.GetElectricFieldAt(pos_x_neg)) / (2 * eps_m);
816 force.x = p.Dot(dE_dx);
817
818 auto pos_y = position_;
819 pos_y.y += eps_m;
820 auto pos_y_neg = position_;
821 pos_y_neg.y -= eps_m;
822 const auto dE_dy = (field_source.GetElectricFieldAt(pos_y) -
823 field_source.GetElectricFieldAt(pos_y_neg)) / (2 * eps_m);
824 force.y = p.Dot(dE_dy);
825
826 auto pos_z = position_;
827 pos_z.z += eps_m;

832 force.z = p.Dot(dE_dz);
833
834 return force;
835 }
