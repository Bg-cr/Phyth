122 [[nodiscard]] Vector3<Quantity<Tesla> >
123 AnalyticSolution(const Vector3<Quantity<Meter> > &point) const {
124 const auto r_vec = point - start_;
125 const auto proj = r_vec.Dot(direction_);
126
127 const auto perp_vec = r_vec - proj * direction_;
128 const auto rho = perp_vec.Length();
129
130 // If point is on the wire axis, field is undefined.
131 // Return zero instead of blowing up.
132 if (rho < Config::epsilon * length_) {
133 return Vector3{0_T, 0_T, 0_T};
134 }
135
136 const auto z1 = -proj;
137 const auto z2 = length_ - proj;
138
139 const auto rho2 = rho * rho;
140 const auto term1 = z2 / Utils::sqrt(rho2 + z2 * z2);
141 const auto term2 = z1 / Utils::sqrt(rho2 + z1 * z1);
142 const auto angular_factor = term1 - term2;
143
144 const auto B_mag = Consts::mu_0 * current_ / (4 * Consts::pi * rho) * angular_factor;
145
146 const auto perp_hat = perp_vec.Normalized();
147 const auto phi_hat = direction_.Cross(perp_hat);
148
149 return B_mag * phi_hat;
150 }
